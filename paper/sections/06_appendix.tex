\section*{Appendix: First-Principles Physical Derivations}
\label{sec:appendix_derivations}

In this appendix, we provide detailed, step-by-step physical derivations from first principles for each equation utilized in the \texttt{hot\_jupiter} physical evolution suite.

\subsection{Derivation A1: 1D Hydrostatic Equilibrium \& Adiabatic Temperature Gradient}
\label{sec:derivation_a1}
Consider a thin spherical fluid shell of radius $r$, thickness $dr$, surface area $A = 4\pi r^2$, and enclosed mass $m(r)$. Radial force balance between inward gravitational attraction and outward hydrostatic pressure gradient requires:
\begin{equation}
A P(r) - A P(r+dr) - \frac{G m(r) \rho(r) A dr}{r^2} = 0 \implies \frac{dP}{dr} = -\frac{G m(r) \rho(r)}{r^2}
\end{equation}
Along a convective adiabat, specific entropy $S$ is constant. Defining the dimensionless adiabatic temperature gradient $\nabla_{\mathrm{ad}} \equiv (\partial \ln T / \partial \ln P)_S$, the interior temperature gradient is:
\begin{equation}
\frac{dT}{dr} = \nabla_{\mathrm{ad}} \frac{T}{P} \frac{dP}{dr} = -\nabla_{\mathrm{ad}} \frac{G m(r) \rho(r) T(r)}{r^2 P(r)}
\end{equation}

\subsection{Derivation A2: Isothermal 3rd-Order Birch-Murnaghan Core EOS}
\label{sec:derivation_a2}
High-pressure heavy-element (rock/ice) cores are modeled using finite strain theory. The Eulerian strain $f$ for a compressed solid sphere of zero-pressure density $\rho_0$ is defined as:
\begin{equation}
f \equiv \frac{1}{2} \left[ \left( \frac{\rho}{\rho_0} \right)^{2/3} - 1 \right]
\end{equation}
Expanding the Helmholtz free energy $F(f)$ to third order in strain $f$ yields the 3rd-order isothermal Birch-Murnaghan equation of state:
\begin{equation}
P(f) = 3 K_0 f (1 + 2f)^{5/2} \left[ 1 + \frac{3}{4}(K_0' - 4) f \right]
\end{equation}
where $K_0 = 200\text{ GPa}$ is the zero-pressure isothermal bulk modulus and $K_0' = 4.0$ is its pressure derivative.

\subsection{Derivation A3: Non-Relativistic Electron Degeneracy Pressure}
\label{sec:derivation_a3}
In deep gas giant envelopes, extreme pressures squeeze hydrogen into a metallic liquid state. According to the Pauli exclusion principle, electrons fill quantum states up to the Fermi momentum $p_F = \hbar (3\pi^2 n_e)^{1/3}$. Integrating momentum space $d^3p$ yields degeneracy pressure:
\begin{equation}
P_{\mathrm{deg}} = \frac{1}{5\pi^2 m_e} \hbar^2 (3\pi^2 n_e)^{5/3} = K_{\mathrm{deg}} \left( \frac{\rho}{\mu_e} \right)^{5/3}
\end{equation}
where $\mu_e = 2/(1+X)$ is the mean molecular weight per electron and $K_{\mathrm{deg}} = 1.08 \times 10^6 \, \text{Pa}\cdot\text{m}^5\cdot\text{kg}^{-5/3}$ is calibrated against SCVH95 tables.

\subsection{Derivation A4: Irradiated Double-Gray Radiative Atmosphere}
\label{sec:derivation_a4}
Semi-analytical irradiated atmosphere boundary conditions follow the Guillot (2010) double-gray formulation. Separating thermal infrared and visible radiation streams, the temperature-optical depth profile $T(\tau)$ is governed by:
\begin{equation}
T^4(\tau) = \frac{3}{4} T_{\mathrm{int}}^4 \left( \tau + \frac{2}{3} \right) + \frac{3}{4} T_{\mathrm{irr}}^4 \left[ \frac{2}{3} + \frac{2}{3\gamma} \left( 1 + \left(\frac{\gamma \tau}{2} - 1\right) e^{-\gamma \tau} \right) \right]
\end{equation}
where $\tau = \kappa_{\mathrm{th}} P / g$, $\gamma \equiv \kappa_{\mathrm{vis}}/\kappa_{\mathrm{th}} \approx 0.1$, $T_{\mathrm{int}}$ is the intrinsic effective temperature, and $T_{\mathrm{irr}} = [(1-A_b) F_{\mathrm{inc}} / (4\sigma_{\mathrm{SB}})]^{1/4}$ is the equilibrium irradiation temperature.

\subsection{Derivation A5: Metallicity-Dependent Core Mass Scaling}
\label{sec:derivation_a5}
In core-accretion planet formation theory, protoplanetary disk solid surface density scales directly with stellar host metallicity $[\mathrm{Fe/H}]$. Heavy-element core mass $M_c$ scales as (Thorngren et al. 2016):
\begin{equation}
M_c([\mathrm{Fe/H}], M_p) = M_{c,0} \left( \frac{M_p}{M_{\mathrm{Jup}}} \right)^{\alpha_M} 10^{\beta_Z [\mathrm{Fe/H}]}
\end{equation}
where $M_{c,0} = 15.0\,M_\oplus$, $\alpha_M = 0.6$, and $\beta_Z = 0.5$.

\subsection{Derivation A6: Quadrupolar Tidal Dissipation Power}
\label{sec:derivation_a6}
A close-in giant planet subject to quadrupolar gravitational tides from its host star experiences cyclic viscoelastic flexing. The time-averaged tidal energy dissipation rate $P_{\mathrm{tidal}}$ deposited into the envelope (Hut 1981) is:
\begin{equation}
P_{\mathrm{tidal}} = \frac{21}{2} \frac{k_2}{Q} \frac{G M_*^2 R_p^5 e^2}{a^6}
\end{equation}
where $k_2 \approx 0.38$ is the planetary Love number of degree 2 and $Q \sim 10^5$ is the tidal quality factor.

\subsection{Derivation A7: MHD Ohmic Dissipation Heating}
\label{sec:derivation_a7}
Atmospheres of strongly irradiated Hot Jupiters ($T_{\mathrm{eq}} > 1200\text{ K}$) thermally ionize alkali metals (Na, K). Zonal winds moving across planetary magnetic fields drive electric currents that dissipate Joule heat into the interior:
\begin{equation}
\begin{aligned}
P_{\mathrm{Ohmic}} &= \int \sigma_{\mathrm{cond}} (\mathbf{v} \times \mathbf{B})^2 dV \\
&\approx \epsilon_{\mathrm{Ohm}} \cdot L_{\mathrm{inc}} \cdot \mathcal{H}(T_{\mathrm{eq}} - 1200\text{ K})
\end{aligned}
\end{equation}
where $\epsilon_{\mathrm{Ohm}} \approx 0.025$ is the Ohmic efficiency factor and $\mathcal{H}$ is the Heaviside step function.

\subsection{Derivation A8: Coupled 6D Orbital \& 3D Spin Vector Dynamics}
\label{sec:derivation_a8}
Simultaneous tidal orbital circularization $\dot{e}$ and semi-major axis decay $\dot{a}$ evolve according to equilibrium tidal theory:
\begin{align}
\frac{\dot{a}}{a} &= -7 \frac{k_2}{Q} \frac{M_*}{M_p} \left(\frac{R_p}{a}\right)^5 e^2, \\
\frac{\dot{e}}{e} &= -\frac{21}{2} \frac{k_2}{Q} \frac{M_*}{M_p} \left(\frac{R_p}{a}\right)^5
\end{align}

\subsection{Derivation A9: Hydrodynamic Roche Lobe Overflow ($L_1$) Mass Loss}
\label{sec:derivation_a9}
When thermal inflation or orbital decay drives planet radius $R_p$ to fill its volume-equivalent Roche lobe radius $R_{\mathrm{Roche}} \approx a [0.49 q^{2/3} / (0.6 q^{2/3} + \ln(1+q^{1/3}))]$, upper envelope gas escapes through the inner Lagrange point $L_1$:
\begin{equation}
\dot{M}_{\mathrm{RLOF}} = -\dot{M}_0 \exp\left[ \eta \left( \frac{R_p - R_{\mathrm{Roche}}}{R_{\mathrm{Roche}}} \right) \right]
\end{equation}
where $\dot{M}_0 \sim 10^{11}\text{ kg/s}$ and $\eta \approx 4$.

\subsection{Derivation A10: Stellar Rotation Tidal Migration}
\label{sec:derivation_a10}
Tidal torque exchange between host star rotation frequency $\Omega_* = 2\pi/P_*$ and planetary orbital mean motion $n = \sqrt{G M_* / a^3}$ drives long-term migration:
\begin{equation}
\left(\frac{\dot{a}}{a}\right)_* = \frac{3 k_{2,*}}{Q_*} \left(\frac{M_p}{M_*}\right) \left(\frac{R_*}{a}\right)^5 (\Omega_* - n)
\end{equation}
Sub-synchronous stars ($n > \Omega_*$) drive inward orbital decay, whereas super-synchronous stars ($n < \Omega_*$) drive outward expansion.

\subsection{Derivation A11: Laplace-Lagrange Secular Perturbation Matrix ($A_{ij}$)}
\label{sec:derivation_a11}
Linearizing Lagrange planetary equations in Poincaré variables $h_i = e_i \sin\varpi_i, k_i = e_i \cos\varpi_i$:
\begin{align}
A_{ii} &= +\frac{n_i}{4} \sum_{j \neq i} \frac{M_j}{M_*} \alpha_{ij} \bar{\alpha}_{ij} b_{3/2}^{(1)}(\alpha_{ij}), \\
A_{ij} &= -\frac{n_i}{4} \frac{M_j}{M_*} \alpha_{ij} \bar{\alpha}_{ij} b_{3/2}^{(2)}(\alpha_{ij})
\end{align}

\subsection{Derivation A12: Observational Selection Function \& SNR Logistic Model}
\label{sec:derivation_a12}
Geometric transit probability $P_{\mathrm{geo}} = \frac{R_* + R_p}{a}$ and Kepler pipeline detection efficiency:
\begin{equation}
P_{\mathrm{det}}(\mathrm{SNR}) = \frac{1}{1 + \exp\left( -\frac{\mathrm{SNR} - 7.1}{1.5} \right)}
\end{equation}
