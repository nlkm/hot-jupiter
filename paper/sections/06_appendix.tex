\section{Step-by-Step Analytical Derivations}
\label{sec:appendix}

In this Appendix, we provide step-by-step mathematical derivations for every primary equation introduced in the text.

\subsection{Derivation A1: 1D Hydrostatic Equations of Equilibrium}
\label{sec:derivation_a1}
Consider a spherically symmetric fluid shell of mass element $dm = 4\pi r^2 \rho \, dr$ located at radius $r$.
1. **Mass Conservation**:
   Rearranging $dm = 4\pi r^2 \rho \, dr$ immediately yields:
   \begin{equation}
   \frac{dr}{dm} = \frac{1}{4\pi r^2 \rho}
   \end{equation}
2. **Momentum Conservation (Hydrostatic Equilibrium)**:
   The net upward pressure force on a radial shell element $[r, r+dr]$ must balance the inward gravitational force exerted by the enclosed mass $m(r)$:
   \begin{equation}
   A \cdot [P(r) - P(r+dr)] = \frac{G m(r) dm}{r^2}
   \end{equation}
   Substituting area $A = 4\pi r^2$ and $P(r+dr) - P(r) = (dP/dr) dr$:
   \begin{equation}
   - 4\pi r^2 \frac{dP}{dr} dr = \frac{G m(r) (4\pi r^2 \rho dr)}{r^2} \implies \frac{dP}{dr} = -\frac{G m(r) \rho(r)}{r^2}
   \end{equation}
   Dividing by $dm/dr = 4\pi r^2 \rho$ yields:
   \begin{equation}
   \frac{dP}{dm} = \frac{dP/dr}{dm/dr} = -\frac{G m}{4\pi r^4}
   \end{equation}
3. **Adiabatic Temperature Gradient**:
   For an isentropic convective envelope, the temperature gradient is governed by the adiabatic index $\nabla_{\mathrm{ad}} \equiv (\partial \ln T / \partial \ln P)_S$:
   \begin{equation}
   \frac{dT}{dm} = \left(\frac{\partial T}{\partial P}\right)_S \frac{dP}{dm} = \nabla_{\mathrm{ad}} \frac{T}{P} \left(-\frac{G m}{4\pi r^4}\right) = -\frac{G m T \nabla_{\mathrm{ad}}}{4\pi r^4 P}
   \end{equation}

\subsection{Derivation A2: Birch-Murnaghan 3rd-Order Isothermal EOS}
\label{sec:derivation_a2}
The Birch-Murnaghan EOS is derived by expanding the Helmholtz free energy $F(f)$ of a compressed solid as a Taylor series in finite Eulerian strain $f$ (Birch 1947):
\begin{equation}
f = \frac{1}{2} \left[ \left(\frac{\rho}{\rho_0}\right)^{2/3} - 1 \right]
\end{equation}
Expanding free energy $F(f)$ to third order in strain $f$:
\begin{equation}
F(f) = a_0 + a_1 f^2 + a_2 f^3 + \mathcal{O}(f^4)
\end{equation}
The isothermal pressure $P$ is the negative volume derivative of free energy:
\begin{equation}
P = -\left(\frac{\partial F}{\partial V}\right)_T = -\left(\frac{\partial F}{\partial f}\right)_T \left(\frac{df}{dV}\right)
\end{equation}
Given $V/V_0 = \rho_0/\rho = (1 + 2f)^{-3/2}$, the volume derivative is:
\begin{equation}
\frac{df}{dV} = -\frac{1}{3 V_0} (1 + 2f)^{5/2} = -\frac{1}{3 V_0} \left(\frac{\rho}{\rho_0}\right)^{5/3}
\end{equation}
Differentiating $F(f) = \frac{9}{2} V_0 K_0 f^2 + \frac{9}{2} V_0 K_0 (K_0' - 4) f^3$:
\begin{equation}
\left(\frac{\partial F}{\partial f}\right)_T = 9 V_0 K_0 f \left[ 1 + \frac{3}{2} (K_0' - 4) f \right]
\end{equation}
Multiplying yields the 3rd-order Birch-Murnaghan equation of state:
\begin{equation}
P(\rho) = \frac{3}{2} K_0 \left[ \left(\frac{\rho}{\rho_0}\right)^{7/3} - \left(\frac{\rho}{\rho_0}\right)^{5/3} \right] \left\{ 1 + \frac{3}{4} (K_0' - 4) \left[ \left(\frac{\rho}{\rho_0}\right)^{2/3} - 1 \right] \right\}
\end{equation}

\subsection{Derivation A3: Non-Relativistic Electron Degeneracy Pressure $K_{\mathrm{DEG}}$}
\label{sec:derivation_a3}
In a fully pressure-ionized electron gas, Pauli's exclusion principle restricts electrons to 2 spin states per phase space cell $h^3$. At zero temperature, electrons fill all momentum states up to Fermi momentum $p_F$:
\begin{equation}
n_e = \frac{N_e}{V} = \frac{2}{h^3} \int_0^{p_F} 4\pi p^2 dp = \frac{8\pi p_F^3}{3 h^3} \implies p_F = \hbar (3\pi^2 n_e)^{1/3}
\end{equation}
The total kinetic energy density $u_{\mathrm{deg}}$ of non-relativistic electrons ($E = p^2 / 2m_e$) is:
\begin{equation}
u_{\mathrm{deg}} = \frac{2}{h^3} \int_0^{p_F} \left(\frac{p^2}{2 m_e}\right) 4\pi p^2 dp = \frac{4\pi p_F^5}{5 m_e h^3} = \frac{\hbar^2 (3\pi^2)^{5/3}}{10 \pi^2 m_e} n_e^{5/3}
\end{equation}
Degeneracy pressure $P_{\mathrm{deg}}$ is related to energy density by $P_{\mathrm{deg}} = \frac{2}{3} u_{\mathrm{deg}}$:
\begin{equation}
P_{\mathrm{deg}} = \frac{(3\pi^2)^{2/3} \hbar^2}{5 m_e} n_e^{5/3}
\end{equation}
Substituting electron number density $n_e = \rho / (\mu_e m_p)$:
\begin{equation}
P_{\mathrm{deg}} = \frac{(3\pi^2)^{2/3} \hbar^2}{5 m_e m_p^{5/3}} \left(\frac{\rho}{\mu_e}\right)^{5/3} = K_{\mathrm{DEG}} \left(\frac{\rho}{\mu_e}\right)^{5/3}
\end{equation}

\subsection{Derivation A4: Guillot (2010) Radiative Transfer Profile}
\label{sec:derivation_a4}
The double-gray radiative transfer equation for an irradiated atmosphere in Eddington approximation is:
\begin{equation}
\frac{d K_{\text{th}}}{d\tau} = \frac{1}{3} \frac{d J_{\text{th}}}{d\tau} = \frac{F_{\text{int}}}{4\pi} + \frac{F_{\text{irr}}}{4\pi} E_2(\gamma \tau)
\end{equation}
Integrating twice over optical depth $\tau$ with boundary condition $J(0) = 2 K(0)$ yields the two-channel thermal temperature profile $T^4(\tau) = \frac{4\pi}{a c} J_{\text{th}}(\tau)$:
\begin{equation}
T^4(\tau) = \frac{3 T_{\mathrm{int}}^4}{4}\left(\tau + \frac{2}{3}\right) + \frac{3 T_{\mathrm{irr}}^4}{4}\left[\frac{2}{3} + \frac{2}{3\gamma}\left(1 + \left(\frac{\gamma \tau}{2} - 1\right)e^{-\gamma \tau}\right)\right]
\end{equation}

\subsection{Derivation A5: Hut (1981) Equilibrium Tidal Dissipation Rates}
\label{sec:derivation_a5}
Equilibrium tidal theory expands the tidal potential $V_{\text{tide}}$ in spherical harmonics $Y_{lm}$ (Kaula 1964). Energy dissipation rate $\dot{E}_{\text{tide}}$ and angular momentum dissipation rate $\dot{h}_{\text{tide}}$ are proportional to the phase lag $\Delta t = (k_2/Q) / (2 n)$:
\begin{equation}
\dot{E}_{\text{tide}} = -\frac{21}{2} \frac{k_2}{Q} \frac{G M_*^2 R_p^5}{a^6} n e^2 - \frac{3}{2} \frac{k_2}{Q} \frac{G M_*^2 R_p^5}{a^6} (\Omega_{\mathrm{rot}} - n \cos\varepsilon)^2
\end{equation}
Relating orbital energy $E = -G M_p M_* / (2a)$ to semi-major axis $a$:
\begin{equation}
\dot{E} = \frac{G M_p M_*}{2 a^2} \frac{da}{dt} \implies \frac{da}{dt} = \frac{2 a^2}{G M_p M_*} \dot{E}_{\text{tide}}
\end{equation}
Substituting $\dot{E}_{\text{tide}}$ yields the semi-major axis decay equation ($da/dt$) and eccentricity damping equation ($de/dt$).
